cmake_minimum_required(VERSION 3.18)
project(issue28090_experiment LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED EXPERIMENT_SOURCE)
    message(FATAL_ERROR "Set EXPERIMENT_SOURCE to the pinned source directory")
endif()
add_subdirectory("${EXPERIMENT_SOURCE}" llama)
add_executable(dense-gate dense_gate.cpp)
target_compile_definitions(dense-gate PRIVATE BACKEND_TEST_SOURCE="${EXPERIMENT_SOURCE}/tests/test-backend-ops.cpp")
target_link_libraries(dense-gate PRIVATE ggml)
set_target_properties(dense-gate PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
